# Undefined-symbol allowlist for the audited libs (tl_foundation_det, tl_sim).
# Spec: docs/CPP-SUBSET.md §4, docs/TESTING.md §5. One glob per line; '#' comments.
# Symbols DEFINED anywhere in the audited set are allowed implicitly ("our own symbols from the
# same lib set") - this file is only for what comes from outside it.
#
# The three sanctioned libc calls (docs/CPP-SUBSET.md §1).
memcpy
memset
memcmp
__memcpy_chk
__memset_chk

# clang-cl / MSVC CRT codegen support: stack probes, cookie checks, and the marker object the
# CRT uses to select the floating-point startup path. These are codegen, not calls we make.
__chkstk
__security_check_cookie
__security_cookie
_fltused
__CxxFrameHandler*
__GSHandlerCheck*

# clang builtins the backend may emit for wide integer math on 32-bit-ish targets. They are
# pure integer routines from compiler-rt, deterministic by definition.
__udivti3
__divti3
__umodti3
__modti3
__muloti4
__aarch64_*
